import React, { useState, useEffect } from 'react';
import { TrendingUp, TrendingDown, Sun, Moon } from 'lucide-react';

// --- 改动 2：模拟 ToolOutput 中的模糊匹配逻辑 ---
// 模糊匹配 key 并取值的辅助函数
const findVal = (obj, keyword) => {
  const key = Object.keys(obj).find(k => k.includes(keyword));
  return key ? parseFloat(obj[key]) : undefined;
};

// --- 改动 1：精简后的 StockQuoteCard 组件 ---
export const StockQuoteCard = ({ data, isDark }) => {
  if (!data) return null;

  // 使用涨跌幅判断正负
  const isPositive = data.changePercent >= 0;
  
  const mainColorClass = isPositive ? 'text-[#eb4335]' : 'text-[#34a853]';
  const bgGradientClass = isPositive 
    ? 'bg-gradient-to-br from-red-50 to-white dark:from-red-950/30 dark:to-gray-900' 
    : 'bg-gradient-to-br from-green-50 to-white dark:from-green-950/30 dark:to-gray-900';
  const badgeColorClass = isPositive
    ? 'bg-red-100 text-red-600 dark:bg-red-900/40 dark:text-red-400'
    : 'bg-green-100 text-green-600 dark:bg-green-900/40 dark:text-green-400';
    
  const TrendIcon = isPositive ? TrendingUp : TrendingDown;

  const DataItem = ({ label, value }) => (
    <div className="flex flex-col space-y-1">
      <span className="text-xs text-gray-500 dark:text-gray-400 font-medium">{label}</span>
      {/* 统一使用默认颜色，去除原有的 colorClass */}
      <span className="text-sm font-mono text-gray-800 dark:text-gray-200 font-medium">{value}</span>
    </div>
  );

  return (
    <div className={`w-full max-w-md bg-white dark:bg-gray-900 rounded-[24px] shadow-xl shadow-gray-200/50 dark:shadow-none border border-gray-100 dark:border-gray-800 overflow-hidden transition-all duration-300 ${isDark ? 'dark' : ''}`}>
      
      <div className={`p-6 pb-5 ${bgGradientClass}`}>
        <div className="flex justify-between items-start mb-4">
          <div>
            <div className="flex items-center space-x-2">
              <h2 className="text-2xl sm:text-3xl font-extrabold text-gray-900 dark:text-white tracking-tight">
                {data.name}
              </h2>
              <span className="px-1.5 py-0.5 rounded text-[10px] font-bold bg-[#1a73e8]/10 text-[#1a73e8] border border-[#1a73e8]/20">融</span>
            </div>
            <div className="flex items-center space-x-2 mt-1.5">
              <span className="text-sm text-gray-500 dark:text-gray-400 font-mono tracking-wider">
                {data.code}
              </span>
            </div>
          </div>
          {/* 已删除状态指示灯 */}
        </div>

        <div className="flex items-end justify-between mt-6">
          <div className="flex items-baseline space-x-4">
            <span className={`text-[3.5rem] leading-none font-black tracking-tighter ${mainColorClass} font-mono`} style={{ fontVariantNumeric: 'tabular-nums' }}>
              {data.price?.toFixed(2)}
            </span>
            <div className="flex flex-col justify-end pb-1 space-y-0.5">
              {/* 已删除涨跌额那一行 */}
              <span className={`text-sm font-bold flex items-center px-1.5 py-0.5 rounded ${badgeColorClass} font-mono`}>
                {isPositive ? '+' : ''}{data.changePercent?.toFixed(2)}%
              </span>
            </div>
          </div>
          <TrendIcon className={`w-12 h-12 mb-2 opacity-15 ${mainColorClass}`} />
        </div>
        {/* 已删除底部时间行 */}
      </div>

      <div className="h-px bg-gradient-to-r from-transparent via-gray-200 dark:via-gray-800 to-transparent"></div>

      <div className="p-6 bg-white dark:bg-gray-900">
        {/* 网格改为 grid-cols-3，只保留三项 */}
        <div className="grid grid-cols-3 gap-y-6 gap-x-4">
          <DataItem label="今开" value={data.open?.toFixed(2) || '--'} />
          <DataItem label="最高" value={data.high?.toFixed(2) || '--'} />
          <DataItem label="最低" value={data.low?.toFixed(2) || '--'} />
        </div>
      </div>
    </div>
  );
};


// --- 主应用展示：模拟解析问财 API 数据并渲染 ---
export default function App() {
  const [isDark, setIsDark] = useState(false);

  // 监听系统主题变化
  useEffect(() => {
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      setIsDark(true);
    }
    window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', e => {
      setIsDark(e.matches);
    });
  }, []);

  const toggleTheme = () => {
    setIsDark(!isDark);
  };

  // 模拟从 iwencai 获取到的原始数据
  const mockIwencaiRawResponse = {
    datas: [{ 
      "股票代码": "600519.SH", 
      "股票简称": "贵州茅台", 
      "最新价": "1281.55", 
      "最新涨跌幅": 0.199, 
      "开盘价_前复权[20260612]": "1271.18", 
      "最高价_前复权[20260612]": "1285.00", 
      "最低价_前复权[20260612]": "1260.00" 
    }]
  };

  let mappedData = null;
  
  // 模拟 ToolOutput 里的 if (isStockQuoteTool(item.name)) 映射逻辑
  if (mockIwencaiRawResponse?.datas?.length > 0) {
    const d = mockIwencaiRawResponse.datas[0];
    mappedData = { 
      name: d['股票简称'], 
      code: d['股票代码'].replace(/\.\w+$/, ''), // 剔除后缀
      price: parseFloat(d['最新价']), 
      changePercent: findVal(d, '涨跌幅'), 
      open: findVal(d, '开盘价'), 
      high: findVal(d, '最高价'), 
      low: findVal(d, '最低价') 
    };
  }

  return (
    <div className={`min-h-screen ${isDark ? 'dark bg-gray-950' : 'bg-gray-50'} flex flex-col items-center justify-center p-4 sm:p-8 font-sans transition-colors duration-300`}>
      <button
        onClick={toggleTheme}
        className={`absolute top-4 right-4 p-2 rounded-full ${isDark ? 'bg-gray-800 text-yellow-400' : 'bg-gray-200 text-gray-800'} transition-colors duration-300`}
        aria-label="Toggle Dark Mode"
      >
        {isDark ? <Sun size={20} /> : <Moon size={20} />}
      </button>
      {mappedData && <StockQuoteCard data={mappedData} isDark={isDark} />}
    </div>
  );
}
